# ==========================================
# 1. CẤU HÌNH MÔI TRƯỜNG (Snowflake CLI)
# ==========================================
$workDir = "C:\SnowflakeAuto"
$sqlPath = "$workDir\copy_into_all.sql"
$logPath = "$workDir\process_log.txt"
# Tên connection khai báo trong file ~/.snowflake/config.toml
$snowConnection = "my_conn" 

# Hàm ghi Log
Function Write-Log {
    Param ([string]$Message)
    $time = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    "$time - $Message" | Out-File -FilePath $logPath -Append
}

Write-Log "========== BẮT ĐẦU QUA TRÌNH NẠP 7 FILE (SNOWFLAKE CLI) =========="

try {
    # ==========================================
    # 2. KHAI BÁO ÁNH XẠ 7 FILE CSV CHỦ LỰC VỚI 7 BẢNG
    # ==========================================
    $tableMapping = @(
        [pscustomobject]@{ CsvPath = "C:\data\nhan_vien.csv"; TableName = "EMPLOYEES_TABLE" },
        [pscustomobject]@{ CsvPath = "C:\data\doanh_thu.csv"; TableName = "REVENUE_TABLE" },
        [pscustomobject]@{ CsvPath = "C:\data\san_pham.csv";  TableName = "PRODUCTS_TABLE" },
        [pscustomobject]@{ CsvPath = "C:\data\ton_kho.csv";   TableName = "INVENTORY_TABLE" },
        [pscustomobject]@{ CsvPath = "C:\data\khach_hang.csv";TableName = "CUSTOMERS_TABLE" },
        [pscustomobject]@{ CsvPath = "C:\data\don_hang.csv";  TableName = "ORDERS_TABLE" },
        [pscustomobject]@{ CsvPath = "C:\data\chi_phi.csv";   TableName = "COSTS_TABLE" }
    )

    $copySqlCommands = @()

    # ==========================================
    # 3. ĐẨY TỪNG FILE LÊN STAGE BẰNG SNOWFLAKE CLI
    # ==========================================
    foreach ($item in $tableMapping) {
        if (Test-Path $item.CsvPath) {
            $csvFile = $item.CsvPath
            $tableName = $item.TableName
            $fileName = [System.IO.Path]::GetFileName($csvFile)

            Write-Log "Đang upload $fileName lên Table Stage (@%$tableName)..."
            
            # Sử dụng lệnh native `snow stage copy` của Snowflake CLI
            $uploadProcess = Start-Process -FilePath "snow" -ArgumentList "stage copy `"$csvFile`" `"@%$tableName`" -c $snowConnection --overwrite" -Wait -NoNewWindow -PassThru

            if ($uploadProcess.ExitCode -eq 0) {
                Write-Log "   -> Tải file $fileName lên Stage thành công."
                
                # Tạo lệnh COPY INTO tương ứng cho bảng
                $copySqlCommands += "COPY INTO $tableName FROM @%$tableName/$fileName FILE_FORMAT = (TYPE = 'CSV' SKIP_HEADER=1 FIELD_OPTIONALLY_ENCLOSED_BY='`"') ON_ERROR = 'CONTINUE';"
            } else {
                Write-Log "   -> LỖI khi upload $fileName! Mã lỗi: $($uploadProcess.ExitCode)"
            }
        } else {
            Write-Log " [CẢNH BÁO] Không tìm thấy file: $($item.CsvPath) -> Bỏ qua bảng $tableName."
        }
    }

    # ==========================================
    # 4. CHẠY LỆNH COPY INTO BẰNG `snow sql`
    # ==========================================
    if ($copySqlCommands.Count -gt 0) {
        $copySqlCommands | Out-File -FilePath $sqlPath -Encoding UTF8
        Write-Log "Đang thực thi lệnh nạp dữ liệu COPY INTO cho các bảng..."

        # Gọi `snow sql` để thực thi file script SQL vừa khởi tạo
        $sqlProcess = Start-Process -FilePath "snow" -ArgumentList "sql -c $snowConnection -f `"$sqlPath`"" -Wait -NoNewWindow -PassThru

        if ($sqlProcess.ExitCode -eq 0) {
            Write-Log "THÀNH CÔNG! Đã nạp dữ liệu hoàn tất vào Snowflake."
        } else {
            Write-Log "THẤT BẠI! Lệnh `snow sql` trả về mã lỗi: $($sqlProcess.ExitCode)."
        }
    } else {
        Write-Log "Không có file CSV nào hợp lệ để xử lý."
    }

} catch {
    Write-Log "LỖI HỆ THỐNG: $($_.Exception.Message)"
}

Write-Log "========== KẾT THÚC QUY TRÌNH ==========`n"
